\section{Methodology}
\label{sec:methodology}

\subsection{Mathematical Definition of N-Queens}
The N-Queens problem requires placing $N$ queens on an $N \times N$ chessboard such that no two queens attack each other. 
In the Boolean paradigm, the problem is modeled using $N^2$ primary decision variables $x_{i,j} \in \{0, 1\}$, where:
\begin{equation}
x_{i,j} =
\begin{cases}
1 & \text{if a queen occupies row } i \text{ and column } j\\
0 & \text{otherwise}
\end{cases}
\end{equation}
for $0 \le i, j < N$.

This formulation strictly enforces four directional constraints:
\begin{itemize}
\item \textbf{Row Constraints:} Exactly one queen per row.
\begin{equation}
\sum_{j=0}^{N-1}x_{i,j}=1 \quad \forall i \in \{0, \dots, N-1\}
\end{equation}
\item \textbf{Column Constraints:} Exactly one queen per column.
\begin{equation}
\sum_{i=0}^{N-1}x_{i,j}=1 \quad \forall j \in \{0, \dots, N-1\}
\end{equation}
\item \textbf{Main Diagonal Constraints:} At most one queen on any main diagonal. Let $d = i - j$. For each $d \in \{-(N-2), \dots, N-2\}$:
\begin{equation}
\sum_{i-j=d}x_{i,j}\le1
\end{equation}
\item \textbf{Anti-Diagonal Constraints:} At most one queen on any anti-diagonal. Let $s = i + j$. For each $s \in \{1, \dots, 2N-3\}$:
\begin{equation}
\sum_{i+j=s}x_{i,j}\le1
\end{equation}
\end{itemize}
Note that diagonals with only a single cell ($d \in \{-(N-1), N-1\}$, $s \in \{0, 2N-2\}$) trivially satisfy the at-most-one condition and do not require explicit constraints.

\subsection{Boolean SAT Constraints and Encodings}
In Boolean Satisfiability, an \textit{Exactly-One} (EO) constraint is represented as the logical conjunction of an \textit{At-Least-One} (ALO) clause and an \textit{At-Most-One} (AMO) constraint:
\begin{equation}
EO(X) = ALO(X) \land AMO(X)
\end{equation}
For a set of boolean variables $X = \{x_1, \ldots, x_m\}$, the ALO constraint is simply a single disjunction:
\begin{equation}
ALO(x_1,\ldots,x_m)=\bigvee_{k=1}^{m}x_k
\end{equation}
The AMO constraint, however, can be encoded into Conjunctive Normal Form (CNF) using various strategies. Formally, an encoding of an At-Most-One constraint over primary literals $X = \{x_1, \ldots, x_m\}$ introduces a set of auxiliary variables $Y = \{y_1, \dots, y_k\}$ and a clause set $E(X, Y)$ satisfying the logical equivalence:
\begin{equation}
\text{AMO}(X) \iff \exists Y.\, E(X, Y)
\end{equation}
Specifically, for any truth assignment $\tau$ on $X$, $\sum_{i=1}^m \tau(x_i) \le 1$ if and only if there exists a truth assignment $\tau'$ on $Y$ such that $\tau \cup \tau'$ satisfies every clause in $E(X, Y)$.

Our framework implements five distinct AMO encoding strategies:
\begin{enumerate}
\item \textbf{Pairwise (Binomial):} Asserts that no two literals in $X$ can simultaneously be true without adding auxiliary variables ($Y = \emptyset$):
\begin{equation}
E_{\text{Pairwise}}(X) = \bigwedge_{1 \le a < b \le m} (\neg x_a \lor \neg x_b)
\end{equation}
Generates $\binom{m}{2} = \frac{m(m-1)}{2}$ binary clauses per group ($\Theta(m^2)$). Across all $6N-6$ constraint groups on the $N \times N$ chessboard, this yields a cubic total clause complexity.
\begin{equation}
\label{eq:pairwise_exact}
C_{\text{pairwise}}(N) = 2N + N^2(N-1) + \frac{N(N-1)(2N-1)}{3} \in \Theta(N^3)
\end{equation}

\item \textbf{Binary (Bitwise):} Introduces $k = \lceil \log_2 m \rceil$ auxiliary variables $Y = \{b_1, \dots, b_k\}$. Each literal $x_i$ ($1 \le i \le m$) is assigned a unique binary bitstring representation $\text{bin}(i-1) = (c_{i,1}, \dots, c_{i,k}) \in \{0, 1\}^k$. Setting $x_i = 1$ forces the auxiliary variables to match this representation:
\begin{equation}
E_{\text{Binary}}(X, Y) = \bigwedge_{i=1}^{m} \bigwedge_{j=1}^{k} \left( \neg x_i \lor l_{i,j} \right), \quad \text{where } l_{i,j} = \begin{cases} b_j & \text{if } c_{i,j} = 1 \\ \neg b_j & \text{if } c_{i,j} = 0 \end{cases}
\end{equation}
This produces $\lceil \log_2 m \rceil$ auxiliary variables and $m \lceil \log_2 m \rceil \in \Theta(m \log m)$ binary clauses per group. Summed over all board rows, columns, and diagonals, the whole-board clause complexity is $\Theta(N^2 \log N)$.

\item \textbf{Sequential Counter:} Implements Sinz's unary counter \citep{sinz2005towards} using $m-1$ auxiliary variables $Y = \{s_1, \dots, s_{m-1}\}$. The variable $s_i$ indicates whether at least one variable in the prefix $\{x_1, \dots, x_i\}$ is true:
\begin{align}
E_{\text{Sequential}}(X, Y) = \; & (\neg x_1 \lor s_1) \land (\neg x_m \lor \neg s_{m-1}) \nonumber \\
& \land \bigwedge_{i=2}^{m-1} \Big( (\neg x_i \lor s_i) \land (\neg s_{i-1} \lor s_i) \land (\neg x_i \lor \neg s_{i-1}) \Big)
\end{align}
Generates $m-1$ auxiliary variables and $3m-4$ clauses for each group of size $m \ge 2$. Across the full board, the exact totals are:
\begin{align}
A_{\text{sequential}}(N) &= 4N^2 - 6N + 2 \in \Theta(N^2) \\
C_{\text{sequential}}(N) &= 12N^2 - 22N + 12 \in \Theta(N^2)
\end{align}

\item \textbf{Commander:} Recursively partitions literals into subgroups of size at most $k=3$ \citep{klieber2007solving}. For each subgroup $G_j \subseteq X$ ($|G_j| \le 3$), a commander variable $c_j$ is introduced, local pairwise exclusion is enforced, and truth propagation is defined via implications:
\begin{equation}
E_{\text{Commander\_group}}(G_j, c_j) = \left( \bigwedge_{g_a, g_b \in G_j, a < b} (\neg g_a \lor \neg g_b) \right) \land \left( \bigwedge_{g \in G_j} (\neg g \lor c_j) \right)
\end{equation}
The set of commander variables $\{c_j\}$ is then recursively constrained by AMO, terminating subdivision whenever a subgroup has size at most $k=3$ where direct pairwise exclusion is applied. This requires $O(m)$ auxiliary variables and clauses per group, yielding $\Theta(N^2)$ clauses across the board.

\item \textbf{2-Product:} Structures literals onto a 2D grid of dimensions $p \times q$ where $p = \lceil \sqrt{m} \rceil$ and $q = \lceil m/p \rceil$ \citep{chen2010new}. Introducing auxiliary row variables $U = \{u_1, \dots, u_p\}$ and column variables $V = \{v_1, \dots, v_q\}$, each literal $x_k$ at grid coordinates $(r, c)$ implies its respective row and column bits:
\begin{equation}
E_{\text{Product}}(X, U, V) = \bigwedge_{k=1}^m \Big( (\neg x_k \lor u_{r(k)}) \land (\neg x_k \lor v_{c(k)}) \Big) \land E_{\text{Pairwise}}(U) \land E_{\text{Pairwise}}(V)
\end{equation}
Requiring $p + q \in O(\sqrt{m})$ auxiliary variables and $2m + \binom{p}{2} + \binom{q}{2} \in O(m)$ clauses per group, this encoding scales to $\Theta(N^2)$ clauses across the board.
\end{enumerate}

The complete algebraic derivations for the exact closed-form expressions \eqref{eq:pairwise_exact} and for Sinz's sequential counter are presented in Appendix~\ref{sec:appendix_proofs}, verified for $N=4$ ($C_{\text{pairwise}}=84, C_{\text{sequential}}=116$) and $N=100$ ($C_{\text{pairwise}}=1,646,900, C_{\text{sequential}}=117,812$).

All SAT formulations are generated via the Python API and passed to the Glucose3 solver \citep{audemard2009predicting} via the PySAT library. To ensure strict isolation of structural efficiency from heuristic tuning, the \texttt{solver\_default} phase policy was enforced.

\subsection{Comparison of Mathematical Formulations and Solving Paradigms}
Our evaluation directly compares three fundamentally different exact solving paradigms applied to the exact same instance sizes of the N-Queens problem.

\textbf{Boolean SAT:} The problem is modeled with $N^2$ primary Boolean variables and purely logical CNF constraints. We evaluate the five AMO encodings (Pairwise, Binary, Sequential, Commander, Product), all solved via the CDCL-based Glucose3 solver with a common phase policy.

\textbf{Constraint Programming (CP):} CP frameworks represent the problem using $N$ integer decision variables $q_i \in \{0, \dots, N-1\}$ (representing the column of the queen in row $i$). The placement constraints are modeled as three global \texttt{AllDifferent} constraints \citep{regin1994filtering}:
\begin{equation}
\text{AllDifferent}(q_0, \dots, q_{N-1})
\end{equation}
\begin{equation}
\text{AllDifferent}(q_0 - 0, \dots, q_{N-1} - (N-1))
\end{equation}
\begin{equation}
\text{AllDifferent}(q_0 + 0, \dots, q_{N-1} + (N-1))
\end{equation}
This formulation was solved independently using OR-Tools CP-SAT and IBM CP Optimizer. While the input formulation is identical, the underlying solving architectures differ fundamentally: IBM CP Optimizer relies on classical CP integer domain filtering, whereas OR-Tools CP-SAT operates as a Lazy Clause Generation (LCG) solver, combining CP propagation with an underlying CDCL Boolean satisfiability core that dynamically derives explanation clauses.

\textbf{Mixed Integer Programming (ILP/MIP):} Using the same $N^2$ binary decision variables $x_{i,j} \in \{0,1\}$ as the SAT model, the MIP formulation translates the logical constraints into continuous linear equality and inequality constraints. For $N \ge 2$, it requires $6N-6$ constraints: $2N$ equality constraints for rows and columns, and $4N-6$ inequality constraints for the diagonals. The model is formulated as a 0-1 integer programming feasibility problem (no objective function) and solved using Gurobi MIP and IBM CPLEX MIP.

\textbf{Experimental Fairness:} To ensure an unbiased comparative baseline, all paradigms are subject to strict experimental fairness parameters. They solve identical N-Queens instance sizes searching for any single valid solution. We enforce single-threaded execution when supported by the solver architecture, apply the same time limits, execute inside the same hardware/software benchmarking environment, and repeat each configuration 5 times. Importantly, all solutions returned by any solver are mathematically verified by a common external validator to guarantee structural correctness. Note that while the formulations are mathematically equivalent, their internal solver representations and search heuristics are entirely distinct.
